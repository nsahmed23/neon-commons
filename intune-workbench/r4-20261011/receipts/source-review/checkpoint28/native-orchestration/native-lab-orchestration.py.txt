class CapacityBlocker(RuntimeError):
    pass


def free_bytes(path):
    value = os.statvfs(path)
    return value.f_bavail * value.f_frsize


def capacity(path, required, phase):
    available = free_bytes(path)
    if available < required:
        raise CapacityBlocker(phase + ': available=' + str(available) + ', required=' + str(required))
    return {'path': str(path), 'available_bytes': available, 'required_bytes': required, 'phase': phase}
BASE_FLAGS = ('--profile', '--profile-sha256', '--project-root', '--evidence-root',
              '--tool-root', '--tool-map', '--scratch-root')


def require(value, message):
    if not value:
        raise ValueError(message)


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=True).encode()


def sha_bytes(raw):
    return hashlib.sha256(raw).hexdigest()


def identity(info):
    return (info.st_dev, info.st_ino, info.st_mode, info.st_uid, info.st_nlink,
            info.st_size, info.st_mtime_ns, info.st_ctime_ns)


def literal(path, expected=None, *, absent=False):
    path = Path(path)
    require(path.is_absolute() and '..' not in path.parts, 'absolute literal path required')
    for ancestor in reversed(path.parents):
        require(stat.S_ISDIR(ancestor.lstat().st_mode), 'literal directory ancestor required')
    if absent:
        require(not path.exists() and not path.is_symlink(), 'fresh unused output required')
    elif expected is not None:
        require(stat.S_IFMT(path.lstat().st_mode) == expected, 'literal input type differs')
    return path


def read_exact(path, maximum=16 * 1024 * 1024):
    path = literal(path, stat.S_IFREG)
    before = path.lstat()
    require(before.st_size <= maximum, 'input exceeds reviewed byte bound')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    with os.fdopen(fd, 'rb') as stream:
        require(identity(before) == identity(os.fstat(stream.fileno())), 'input open changed')
        raw = stream.read(maximum + 1)
        require(len(raw) <= maximum and identity(before) == identity(os.fstat(stream.fileno())), 'input read changed')
    require(identity(before) == identity(path.lstat()), 'input namespace changed')
    return raw


def json_exact(raw):
    def pairs(rows):
        result = {}
        for key, value in rows:
            require(key not in result, 'duplicate JSON key')
            result[key] = value
        return result
    def invalid(value):
        raise ValueError('nonfinite JSON number: ' + value)
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=invalid)


def verify_pin(value, maximum=16 * 1024 * 1024):
    require(type(value) is dict and set(value) == {'path', 'bytes', 'sha256'}, 'exact artifact pin fields required')
    require(type(value['bytes']) is int and 0 <= value['bytes'] <= maximum, 'artifact byte pin invalid')
    require(type(value['sha256']) is str and re.fullmatch('[0-9a-f]{64}', value['sha256']), 'artifact digest pin invalid')
    raw = read_exact(value['path'], maximum)
    require(len(raw) == value['bytes'] and sha_bytes(raw) == value['sha256'], 'artifact exact bytes differ: ' + value['path'])
    return raw


def write_new(path, value):
    raw = canonical(value) + b'\n'
    with Path(path).open('xb') as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())
    fd = os.open(Path(path).parent, os.O_RDONLY | os.O_DIRECTORY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)

def validate_contract(arguments):
    config_path = literal(arguments.configuration, stat.S_IFREG)
    config_raw = read_exact(config_path)
    require(sha_bytes(config_raw) == arguments.configuration_sha256, 'configuration byte pin differs')
    config = json_exact(config_raw)
    fields = {'schema_version', 'source_revision', 'prepared_receipt', 'base_command', 'profile',
              'tool_map', 'daemon_readiness', 'run_root', 'durable_root', 'plan_sha256', 'limits'}
    require(type(config) is dict and set(config) == fields and config['schema_version'] == 'c25-native-outer-orchestration/1', 'closed configuration fields required')
    require(config['source_revision'] == C25, 'only exact C25 source is admitted')
    prepared = json_exact(verify_pin(config['prepared_receipt']))
    require(prepared.get('status') == 'PREPARED_NOT_NATIVE_QUALIFIED' and prepared.get('source_commit') == C25
            and prepared.get('native_dispatches') == 0 and prepared.get('package_bytes_unchanged') is True
            and prepared.get('both_initial_extractions_match_exact_archives') is True, 'exact prepared source contract required')
    base = json_exact(verify_pin(config['base_command']))
    profile_raw = verify_pin(config['profile'])
    profile = json_exact(profile_raw)
    verify_pin(config['tool_map'])
    daemon = json_exact(verify_pin(config['daemon_readiness']))
    verify_pin(daemon['worker_host_source'])
    require(daemon.get('schema_version') == 'c25-native-explicit-daemon-readiness/1'
            and daemon.get('status') == 'PASS_READ_ONLY_EXPLICIT_WORKER_ROUTE'
            and daemon.get('source_revision') == C25 and daemon.get('docker') == prepared['tools']['docker']
            and daemon.get('native_dispatches') == 0 and daemon.get('container_mutations') == 0
            and daemon.get('endpoint') == 'unix:///var/run/docker.sock'
            and daemon.get('socket', {}).get('path') == '/var/run/docker.sock'
            and daemon.get('socket', {}).get('resolved_path') == '/run/docker.sock'
            and daemon.get('client_config_sha256') == sha_bytes(b'{"auths":{}}\n')
            and daemon.get('client_environment') == {'PATH': '/usr/bin:/bin', 'LANG': 'C.UTF-8', 'LC_ALL': 'C.UTF-8'},
            'read-only same-env explicit worker endpoint binding required')
    require(profile.get('source_revision') == C25 and profile.get('loader_sha256') == LOADER_SHA, 'profile/loader source differs')
    require(type(base) is list and len(base) == 18 and all(type(item) is str for item in base)
            and base[1:3] == ['-B', '-I'] and tuple(base[4::2]) == BASE_FLAGS, 'frozen literal base argv required')

    require(daemon['worker_host_source']['path'] == str(Path(options['--project-root']) / 'intune_iac' / 'worker_host.py'),
            'read-only endpoint evidence is not bound to the actual prepared worker source')
    require(options['--profile'] == config['profile']['path'] and options['--profile-sha256'] == config['profile']['sha256']
            and options['--tool-map'] == config['tool_map']['path'], 'base does not bind selected profile/map')
    require(prepared['profile'] == {key: config['profile'][key] for key in ('bytes', 'sha256')}
            and prepared['base_command'] == {key: config['base_command'][key] for key in ('bytes', 'sha256')}, 'prepared profile/base pins differ')
    require(sha_bytes(read_exact(base[3])) == LOADER_SHA and prepared['loader_pin']['sha256'] == LOADER_SHA,
            'literal frozen loader differs')
    require(Path(base[0]).resolve() == Path(prepared['tools']['python']['path']), 'base interpreter identity differs')
    verify_pin(prepared['tools']['python'], 512 * 1024 * 1024)
    profile_root = literal(prepared['out'], stat.S_IFDIR)
    scratch = literal(prepared['scratch'], stat.S_IFDIR)
    require(options['--scratch-root'] == str(scratch) and Path(options['--profile']).parent == profile_root
            and Path(base[3]).parent.parent == profile_root and Path(options['--evidence-root']).parent == profile_root,

    run_root = literal(config['run_root'], absent=True)

    require(run_root != durable, 'journey and durable roots must be distinct fresh namespaces')

    roots = [run_root, durable, *protected]

            and 32 * 1024**2 <= limits['workspace_reserve_bytes'] <= 1024**3
            and 768 * 1024**2 <= limits['per_action_reservation_bytes'] <= 2 * 1024**3,

            and admission['schema_version'] == 'c25-native-outer-orchestration-admission/1'
            and admission['status'] == 'APPROVED_LOCAL_LAB_EXECUTION'
            and admission['configuration_sha256'] == arguments.configuration_sha256
            and admission['driver_sha256'] == sha_bytes(read_exact(Path(__file__).absolute()))
            and admission['profile_sha256'] == config['profile']['sha256']
            and admission['base_command_sha256'] == config['base_command']['sha256']
            and admission['source_revision'] == C25 and admission['scope'] == SCOPE
            and type(admission['decision_owner']) is str and admission['decision_owner']
            and type(admission['expires_at']) is int and time.time() < admission['expires_at']
            and admission['coordinator_go_sha256'] == sha_bytes(arguments.coordinator_go.encode()),
            'reviewed local admission or explicit coordinator GO differs')
    return config, prepared, base, options, admission, config_raw, admission_raw


def check_worker_daemon(config, prepared, output):
    """Read-only check of the actual receiving route used by frozen worker_host."""
    expected = json_exact(verify_pin(config['daemon_readiness']))
    verify_pin(expected['worker_host_source'])
    verify_pin(prepared['tools']['docker'], 512 * 1024 * 1024)
    socket = Path(expected['socket']['path']); observed_socket = socket.lstat()
    wanted = expected['socket']
    require(stat.S_ISSOCK(observed_socket.st_mode) and not socket.is_symlink()
            and str(socket.resolve(strict=True)) == wanted['resolved_path']
            and all(getattr(observed_socket, 'st_' + key) == wanted[key] for key in ('dev', 'ino', 'uid', 'gid', 'mode')),
            'explicit worker Docker socket identity or canonical alias changed before dispatch')
    client_config = output.parent / 'daemon-client-config'
    client_config.mkdir(mode=0o700)
    client_config_bytes = b'{"auths":{}}\n'
    with (client_config / 'config.json').open('xb') as stream:
        stream.write(client_config_bytes)
        stream.flush()
        os.fsync(stream.fileno())
    require(sha_bytes(read_exact(client_config / 'config.json')) == expected['client_config_sha256'],
            'private read-only Docker client configuration differs from actual worker schema')
    fmt = '{"ID":{{json .ID}},"ServerVersion":{{json .ServerVersion}},"Name":{{json .Name}}}'
    argv = [prepared['tools']['docker']['path'], '--config', str(client_config), '--host', expected['endpoint'],
            'info', '--format', fmt]
    process = subprocess.run(argv, capture_output=True, env=expected['client_environment'], timeout=20)
    record = {'schema_version': 'c25-native-explicit-daemon-recheck/1', 'argv': argv,
              'exit_code': process.returncode, 'stdout_base64': base64.b64encode(process.stdout).decode(),
              'stderr_base64': base64.b64encode(process.stderr).decode(),
              'expected_receiving_identity': expected['server_identity'], 'endpoint': expected['endpoint'],
              'client_config_sha256': expected['client_config_sha256'], 'native_dispatches': 0, 'container_mutations': 0}
    write_new(output, record)
    require(process.returncode == 0 and json_exact(process.stdout) == expected['server_identity'],
            'actual explicit worker Docker receiving identity differs before native dispatch')
    return {'status': 'PASS_READ_ONLY_EXPLICIT_WORKER_DAEMON_RECHECK', 'record_path': str(output),
            'record_sha256': sha_bytes(read_exact(output)), 'native_dispatches': 0}


class Journal:
    def __init__(self, root):
        self.path = root / 'JOURNAL.jsonl'
        self.stream = self.path.open('xb')
        self.previous = None
        self.count = 0
    def append(self, kind, **facts):
        self.count += 1
        event = {'schema_version': 'c25-native-outer-event/1', 'sequence': self.count,
                 'time_ns': time.time_ns(), 'kind': kind, 'previous_sha256': self.previous, **facts}
        raw = canonical(event)
        self.stream.write(raw + b'\n'); self.stream.flush(); os.fsync(self.stream.fileno())
        self.previous = sha_bytes(raw)
    def close(self):
        self.stream.close()


def action_argv(base, row, run_root):
    result = list(base) + ['--action', row['action'], '--output', str(run_root / row['output'])]
    for key, value in row['arguments'].items():
        flag = '--' + key.replace('_', '-')
        if key == 'cancel_after_acquire':
            require(value is True, 'only deliberate cancellation is declared')
            result.append(flag)
        else:
            result += [flag, str(run_root / value)]
    return result


def stop_child(process, limits):
    """Allow the actual qualifier finally blocks; never silently continue mutation."""
    events = []
    for selected, grace in ((signal.SIGINT, limits['interrupt_grace_seconds']),
                            (signal.SIGTERM, limits['terminate_grace_seconds'])):
        if process.poll() is not None:
            break
        try:
            os.killpg(process.pid, selected)
            events.append({'signal': selected.name, 'process_group': process.pid})
        except ProcessLookupError:
            break
        try:
            process.wait(timeout=grace)
        except subprocess.TimeoutExpired:
            pass
    if process.poll() is None:
        os.killpg(process.pid, signal.SIGKILL)
        events.append({'signal': 'SIGKILL', 'process_group': process.pid, 'outcome_uncertain': True})
        process.wait(timeout=10)
    return events


def inspect_portable(row, output, config, base_options):
    receipt_path = output / 'PORTABLE-QUALIFICATION.json'
    raw = read_exact(receipt_path)
    value = json_exact(raw)
    require(value.get('schema_version') == 'portable-native-adoption-qualification-receipt/1'
            and value.get('status') == 'PASS' and value.get('action') == row['action']
            and value.get('source_revision') == C25 and value.get('profile_sha256') == config['profile']['sha256']
            and value.get('loader_sha256') == LOADER_SHA and value.get('project_root') == base_options['--project-root']
            and value.get('native_platform_qualified') is False and value.get('production_approved') is False
            and type(value.get('scenario_result')) is dict and value['scenario_result'].get('status') == 'PASS',
            'exact portable action receipt did not pass scoped contract')
    for name in ('FINAL-QUALIFICATION.json', 'cleanup.json', 'FINAL-CLEANUP.json', 'FIXTURE-CLEANUP.json'):
        path = output / name
        if path.is_symlink() or path.exists():
            cleanup = json_exact(read_exact(path))
            require(type(cleanup) is dict, 'scenario cleanup receipt must be a typed object: ' + name)
            if name == 'FINAL-QUALIFICATION.json':
                required_lists = [key for key in ('cleanup', 'cleanup_failures') if key in cleanup]
                require(cleanup.get('status') == 'PASS' and bool(required_lists)
                        and 'primary_error' in cleanup and cleanup['primary_error'] is None
                        and cleanup.get('platform_qualified') is False
                        and ('assertion_receipt_status' not in cleanup or cleanup['assertion_receipt_status'] == 'PASS')
                        and ('production_approved' not in cleanup or cleanup['production_approved'] is False),
                        'final qualification lacks explicit completed scoped proof: ' + name)
            else:
                require('failures' in cleanup and ('status' not in cleanup or cleanup['status'] == 'PASS'),
                        'scenario cleanup lacks its actual typed failure-list proof: ' + name)
            lists = [cleanup[key] for key in ('cleanup', 'failures', 'cleanup_failures') if key in cleanup]
            require(all(type(items) is list and not items for items in lists)
                    and ('thread_alive' not in cleanup or cleanup['thread_alive'] is False),
                    'scenario cleanup remains incomplete: ' + name)
    return {'path': str(receipt_path), 'bytes': len(raw), 'sha256': sha_bytes(raw),
            'scenario_check_count': len(value['scenario_result'].get('checks', {})),
            'check_counts_overlap_and_are_not_added_to_strict_totals': True}


def census(roots, limits):
    rows = {}
    total = 0
    def visit(path, name):
        nonlocal total
        require(len(rows) < limits['max_entries'], 'capsule entry bound reached')
        info = path.lstat()
        row = {'source_path': str(path), 'mode': info.st_mode, 'identity': list(identity(info))}
        if stat.S_ISDIR(info.st_mode):
            row['type'] = 'directory'; rows[name] = row
            for child in sorted(path.iterdir()):
                visit(child, name + '/' + child.name)
            require(identity(info) == identity(path.lstat()), 'directory changed during census')
        elif stat.S_ISREG(info.st_mode):
            require(info.st_size <= limits['max_file_bytes'], 'capsule file exceeds bound')
            total += info.st_size; require(total <= limits['max_total_bytes'], 'capsule total bound reached')
            fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
            with os.fdopen(fd, 'rb') as stream:
                require(identity(info) == identity(os.fstat(stream.fileno())), 'capsule file open changed')
                digest = hashlib.file_digest(stream, 'sha256').hexdigest()
                require(identity(info) == identity(os.fstat(stream.fileno())), 'capsule file read changed')
            require(identity(info) == identity(path.lstat()), 'capsule file namespace changed')
            rows[name] = {**row, 'type': 'file', 'bytes': info.st_size, 'sha256': digest}
        elif stat.S_ISLNK(info.st_mode):
            target = os.readlink(os.fsencode(path))
            require(identity(info) == identity(path.lstat()), 'literal link changed')
            rows[name] = {**row, 'type': 'literal_link', 'target_base64': base64.b64encode(target).decode(), 'bytes': len(target)}
        elif stat.S_ISSOCK(info.st_mode):
            # Closed brokers deliberately retain their Unix socket namespace.
            # A socket has no file payload: preserve type/mode/identity only,
            # never connect, follow, serialize peer memory or reconstruct one.
            rows[name] = {**row, 'type': 'runtime_socket_metadata', 'bytes': 0,
                          'connection_state': 'not inferred from namespace; no live state serialization'}
        else:
            raise ValueError('Unsupported special node preserved in raw tree; capsule closure unavailable: ' + str(path))
    for label, path in roots.items():
        require(re.fullmatch('[a-z][a-z0-9_-]*', label), 'capsule namespace label invalid')
        visit(literal(path, stat.S_IFDIR), label)
    return {'entries': dict(sorted(rows.items())), 'total_regular_file_bytes': total}


def capsule(roots, capsule_root, durable, limits):
    before = census(roots, limits)
    write_new(capsule_root / 'CAPSULE-CENSUS-BEFORE.json', before)
    # Worst-case complete raw bytes plus conservative ZIP/member overhead.
    # Compression is not a license to assume capacity before measuring it.
    archive_reservation = before['total_regular_file_bytes'] + len(before['entries']) * 4096 + limits['temporary_reserve_bytes']
    capacity(capsule_root, archive_reservation, 'temporary complete-capsule admission')
    archive = capsule_root / 'PRIVATE-RAW-CAPSULE.zip'
    with zipfile.ZipFile(archive, 'x', compression=zipfile.ZIP_DEFLATED, allowZip64=True) as packed:
        for name, row in before['entries'].items():
            rel = PurePosixPath(name)
            require(not rel.is_absolute() and all(p not in ('', '.', '..') for p in rel.parts)
                    and '\\' not in name and ':' not in name, 'unsafe ZIP name retained in raw tree; cannot seal')
            directory = row['type'] == 'directory'
            item = zipfile.ZipInfo(name + ('/' if directory else ''), date_time=(1980, 1, 1, 0, 0, 0))
            item.create_system = 3; item.external_attr = row['mode'] << 16
            item.compress_type = zipfile.ZIP_DEFLATED
            if directory:
                item.external_attr |= 0x10
                packed.writestr(item, b'')
            elif row['type'] == 'runtime_socket_metadata':
                packed.writestr(item, b'')
            elif row['type'] == 'literal_link':
                packed.writestr(item, base64.b64decode(row['target_base64'], validate=True))
            else:
                path = Path(row['source_path'])
                fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
                with os.fdopen(fd, 'rb') as source, packed.open(item, 'w', force_zip64=True) as target:
                    require(list(identity(os.fstat(source.fileno()))) == row['identity'], 'archive input changed before copy')
                    digest = hashlib.sha256(); size = 0
                    while True:
                        raw = source.read(1024 * 1024)
                        if not raw:
                            break
                        target.write(raw); digest.update(raw); size += len(raw)
                    require(size == row['bytes'] and digest.hexdigest() == row['sha256']
                            and list(identity(os.fstat(source.fileno()))) == row['identity'], 'archive member bytes changed')
    observed = {}
    with zipfile.ZipFile(archive) as packed:
        for item in packed.infolist():
            name = item.filename.rstrip('/') if item.is_dir() else item.filename
            require(name not in observed and name in before['entries'], 'duplicate or extra capsule member')
            row = before['entries'][name]
            require(stat.S_IFMT(item.external_attr >> 16) == stat.S_IFMT(row['mode']), 'capsule member type differs')
            with packed.open(item) as stream:
                if row['type'] == 'literal_link':
                    require(stream.read() == base64.b64decode(row['target_base64'], validate=True), 'literal link bytes differ')
                elif row['type'] == 'file':
                    require(item.file_size == row['bytes'] and hashlib.file_digest(stream, 'sha256').hexdigest() == row['sha256'], 'capsule readback bytes differ')
                else:
                    require(stream.read() == b'', 'directory payload differs')
            observed[name] = True
    require(set(observed) == set(before['entries']), 'capsule members missing')
    after = census(roots, limits)
    write_new(capsule_root / 'CAPSULE-CENSUS-AFTER.json', after)
    require(before == after, 'raw trees changed during sealing; no closure')
    with archive.open('rb') as stream:
        digest = hashlib.file_digest(stream, 'sha256').hexdigest()
    temporary_pin = {'path': str(archive), 'bytes': archive.stat().st_size, 'sha256': digest}
    write_new(capsule_root / 'TEMPORARY-CAPSULE-READBACK.json', {
        'status': 'PASS_EXACT_TYPED_PRIVATE_TEMPORARY_CAPSULE', 'artifact': temporary_pin,
        'member_count': len(before['entries']), 'raw_tree_before_after_equal': True,
        'runnable_restore': False, 'runtime_socket_nodes_are_metadata_only': True})
    durable_target = durable / 'PRIVATE-RAW-CAPSULE.zip'
    delivery = {'status': 'BLOCKED_DURABLE_CAPSULE_CAPACITY', 'measured_available_bytes': free_bytes(durable),
                'required_bytes': temporary_pin['bytes'] + limits['workspace_reserve_bytes'],
                'temporary_complete_capsule_preserved': temporary_pin}
    if delivery['measured_available_bytes'] >= delivery['required_bytes']:
        partial = durable / 'PRIVATE-RAW-CAPSULE.zip.partial'
        try:
            copied = hashlib.sha256()
            with archive.open('rb') as source, partial.open('xb') as target:
                while True:
                    raw = source.read(1024 * 1024)
                    if not raw:
                        break
                    capacity(durable, len(raw) + limits['workspace_reserve_bytes'], 'durable capsule copy')
                    target.write(raw); copied.update(raw)
                target.flush(); os.fsync(target.fileno())
            require(partial.stat().st_size == temporary_pin['bytes'] and copied.hexdigest() == digest, 'durable capsule copy bytes differ')
            with partial.open('rb') as stream:
                require(hashlib.file_digest(stream, 'sha256').hexdigest() == digest, 'durable capsule saved readback differs')
            require(not durable_target.exists(), 'never overwrite durable capsule')
            partial.rename(durable_target)
            delivery = {'status': 'PASS_EXACT_DURABLE_PRIVATE_CAPSULE', 'path': str(durable_target),
                        'bytes': temporary_pin['bytes'], 'sha256': digest}
        except BaseException as error:
            delivery = {'status': 'BLOCKED_DURABLE_CAPSULE_COPY', 'error_type': type(error).__name__,
                        'error': str(error), 'partial_copy_preserved': str(partial),
                        'temporary_complete_capsule_preserved': temporary_pin, 'automatic_retry': False}
    return {'status': 'PASS_EXACT_TYPED_PRIVATE_TEMPORARY_CAPSULE', 'path': str(archive), 'bytes': archive.stat().st_size,
            'durable_delivery': delivery,
            'sha256': digest, 'entry_count': len(before['entries']), 'literal_links_preserved_without_following':
            sum(row['type'] == 'literal_link' for row in before['entries'].values()),
            'roots': {key: str(value) for key, value in roots.items()},
            'runtime_socket_metadata_count': sum(row['type'] == 'runtime_socket_metadata' for row in before['entries'].values()),
            'runnable_restore': False,
            'omissions': ['Existing source/runtime packages and external native tool binaries remain in their separately pinned preserved locations, not copied into this raw capsule.',
                          'Capsule, its external census/receipt and subsequent final status cannot contain their own final bytes; they remain alongside it.',
                          'Runtime Unix sockets are explicit metadata-only typed markers: no live connection/peer state is serializable or restored. Every regular tool/source copy inside selected raw trees remains included byte-exact; no content deduplication or pruning occurs.'],
            'public_suitable': False, 'historical_raw_loss_closed': False}


def run(arguments):
    require(sys.dont_write_bytecode and sys.flags.isolated, 'invoke this driver with Python -I -B')
    os.umask(0o077)
    config, prepared, base, options, admission, config_raw, admission_raw = validate_contract(arguments)
    run_root = Path(config['run_root']); durable = Path(config['durable_root'])
    run_root.mkdir(mode=0o700); durable.mkdir(mode=0o700)
    journey = run_root / 'journey'; journey.mkdir(mode=0o700)
    control = run_root / 'control'; control.mkdir(mode=0o700)
    capsule_root = run_root / 'capsule'; capsule_root.mkdir(mode=0o700)
    (control / 'configuration.json').write_bytes(config_raw)
    (control / 'admission.json').write_bytes(admission_raw)
    (control / 'daemon-readiness.json').write_bytes(verify_pin(config['daemon_readiness']))
    write_new(control / 'ACTION-PLAN.json', make_plan())
    journal = Journal(control)
    journal.append('authorized_lab_start', source_revision=C25, configuration_sha256=arguments.configuration_sha256,
                   admission_sha256=arguments.admission_sha256, profile_sha256=config['profile']['sha256'], scope=SCOPE)
    completed = []
    primary = None
    result = {'schema_version': 'c25-native-outer-orchestration-result/1', 'status': 'STARTED',
              'source_revision': C25, 'profile_sha256': config['profile']['sha256'],
              'configuration_sha256': arguments.configuration_sha256,
              'scope': SCOPE, 'platform_qualified': False, 'production_approved': False,
              'historical_raw_loss_closed': False}
    def termination(signum, frame):
        raise KeyboardInterrupt('Outer termination requested: ' + str(signum))
    prior_handler = signal.signal(signal.SIGTERM, termination)
    try:
        for index, row in enumerate(make_plan(), 1):
            require(time.time() < admission['expires_at'], 'coordinator admission expired before dispatch')
            require(set(row['depends_on']) <= set(completed), 'ordered predecessor not qualified')
            verify_pin(config['profile']); verify_pin(config['base_command'])
            require(sha_bytes(read_exact(base[3])) == LOADER_SHA, 'frozen loader changed before dispatch')
            output = journey / row['output']
            require(not output.exists() and not output.is_symlink(), 'never requeue or overwrite a prior operation')
            command_root = control / ('command-' + str(index).zfill(3) + '-' + row['id'])
            command_root.mkdir(mode=0o700)
            argv = action_argv(base, row, journey)
            write_new(command_root / 'COMMAND.json', {'argv': argv, 'action_id': row['id'], 'output': str(output)})
            admitted_capacity = capacity(run_root, config['limits']['temporary_reserve_bytes'] + config['limits']['per_action_reservation_bytes'], 'before action ' + row['id'])
            daemon_check = check_worker_daemon(config, prepared, command_root / 'DAEMON-READINESS.json')
            journal.append('dispatch_started', action_id=row['id'], action=row['action'], argv=argv, capacity=admitted_capacity, daemon=daemon_check,
                           output=str(output), expected_qualification_status='PASS',
                           known_bad_product_outcomes_remain_negative=True)
            process = None
            with (command_root / 'stdout.bin').open('xb') as stdout, (command_root / 'stderr.bin').open('xb') as stderr:
                try:
                    process = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=stdout, stderr=stderr,
                        cwd=options['--project-root'], start_new_session=True,
                        env={'PATH': '/usr/bin:/bin', 'LANG': 'C.UTF-8', 'PYTHONDONTWRITEBYTECODE': '1', 'PYTHONNOUSERSITE': '1'})
                    deadline = time.monotonic() + config['limits']['action_timeout_seconds']
                    while True:
                        capacity(run_root, config['limits']['temporary_reserve_bytes'], 'in-flight action ' + row['id'])
                        remaining = deadline - time.monotonic()
                        if remaining <= 0:
                            raise subprocess.TimeoutExpired(argv, config['limits']['action_timeout_seconds'])
                        try:
                            code = process.wait(timeout=min(0.25, remaining))
                            break
                        except subprocess.TimeoutExpired:
                            pass
                except BaseException as error:
                    cleanup = []
                    try:
                        cleanup = stop_child(process, config['limits']) if process is not None else []
                    except BaseException as secondary:
                        cleanup.append({'error_type': type(secondary).__name__, 'error': str(secondary), 'outcome_uncertain': True})
                        error.add_note('Secondary child cleanup: ' + str(secondary))
                    try:
                        journal.append('dispatch_interrupted', action_id=row['id'], error_type=type(error).__name__,
                                       process_cleanup=cleanup, operation_outcome='uncertain', automatic_retry=False)
                    except BaseException as secondary:
                        error.add_note('Secondary interruption journaling: ' + str(secondary))
                    raise
                finally:
                    active = sys.exception()
                    for stream in (stdout, stderr):
                        try:
                            stream.flush(); os.fsync(stream.fileno())
                        except BaseException as secondary:
                            if active is None:
                                raise
                            active.add_note('Secondary raw log flush: ' + str(secondary))
            write_new(command_root / 'EXIT.json', {'exit_code': code, 'action_id': row['id']})
            require(code == 0, 'unexpected loader failure: ' + row['id'])
            receipt = inspect_portable(row, output, config, options)
            write_new(command_root / 'QUALIFIED.json', receipt)
            completed.append(row['id'])
            journal.append('dispatch_qualified', action_id=row['id'], receipt=receipt,
                           completed_action_count=len(completed), temporary_available_bytes=free_bytes(run_root),
                           workspace_available_bytes=free_bytes(durable), overlapping_check_counts_not_added=True)
            if row['id'] == 'adoption':
                measured = census({'journey_raw': journey, 'scratch_raw': Path(prepared['scratch'])}, config['limits'])
                write_new(control / 'FIRST-ADOPTION-CAPACITY.json', {
                    'source': 'actual no-follow owned-tree census after first adopted fixture',
                    'measured_raw_bytes': measured['total_regular_file_bytes'], 'entry_count': len(measured['entries']),
                    'remaining_action_count': len(make_plan()) - len(completed),
                    'projection_not_a_capacity_or_performance_pass': True,
                    'linear_upper_planning_example_bytes': measured['total_regular_file_bytes'] * (len(make_plan()) - len(completed)),
                    'temporary_available_bytes': free_bytes(run_root), 'workspace_available_bytes': free_bytes(durable),
                    'native_tool_aliases_or_copies_preserved': [name for name, entry in measured['entries'].items()
                        if entry['type'] == 'file' and any(entry['sha256'] == tool['sha256'] for tool in prepared['tools'].values())]})
                # Freeze a complete current evidence checkpoint before the next
                # native action. This is a measured capacity sample, never a
                # promise about fleet performance or remaining archive size.
                first_capsule = capsule_root / 'after-adoption'; first_capsule.mkdir(mode=0o700)
                first_durable = durable / 'after-adoption'; first_durable.mkdir(mode=0o700)
                journal.append('first_adoption_preservation_started', native_dispatch_paused_at_safe_between_action_boundary=True)
                first = capsule({'journey_raw': journey, 'scratch_raw': Path(prepared['scratch']),
                                 'profile_inputs': Path(prepared['out']), 'outer_control': control},
                                first_capsule, first_durable, config['limits'])
                write_new(first_capsule / 'CAPSULE-RECEIPT.json', first)
                write_new(first_durable / 'CAPSULE-RECEIPT.json', first)
                journal.append('first_adoption_actual_capacity_sample', temporary_capsule_bytes=first['bytes'],
                               temporary_available_bytes=free_bytes(run_root), workspace_available_bytes=free_bytes(durable),
                               durable_delivery=first['durable_delivery'], further_capacity_checked_per_action=True,
                               sample_is_not_remaining_workload_capacity_or_fleet_qualification=True)
                if first['durable_delivery']['status'] != 'PASS_EXACT_DURABLE_PRIVATE_CAPSULE':
                    raise CapacityBlocker('First adopted fixture capsule cannot be delivered durably; stop before further native dispatch')
        result['status'] = 'PASS_SCOPED_NATIVE_LAB_JOURNEYS'
    except BaseException as error:
        primary = error
        result.update(status='BLOCKED_CAPACITY' if isinstance(error, CapacityBlocker) else 'STOPPED_UNEXPECTED_FAILURE', error_type=type(error).__name__, error=str(error), automatic_retry=False)
        try:
            journal.append('stopped_unexpected_failure', error_type=type(error).__name__, error=str(error),
                           completed_action_ids=completed, automatic_retry=False, native_outcome_may_be_uncertain=True)
        except BaseException as secondary:
            error.add_note('Secondary stopped-event journaling: ' + str(secondary))
    finally:
        def finalization_failure(stage, error):
            nonlocal primary
            result.setdefault('finalization_errors', []).append({
                'stage': stage, 'error_type': type(error).__name__, 'error': str(error)})
            if primary is not None:
                primary.add_note('Secondary ' + stage + ': ' + str(error))
            else:
                primary = error
                result['status'] = 'FAIL_REQUIRED_FINALIZATION'
        try:
            signal.signal(signal.SIGTERM, prior_handler)
        except BaseException as error:
            finalization_failure('signal_restore', error)
        result['completed_action_ids'] = completed
        result['planned_action_count'] = len(make_plan())
        result['not_started_action_ids'] = [row['id'] for row in make_plan() if row['id'] not in completed]
        try:
            journal.append('raw_preservation_started', scope=SCOPE, completed_action_ids=completed)
        except BaseException as error:
            finalization_failure('raw_preservation_journal', error)
        try:
            journal.close()
        except BaseException as error:
            finalization_failure('journal_close', error)
        roots = {'journey_raw': journey, 'scratch_raw': Path(prepared['scratch']),
                 'profile_inputs': Path(prepared['out']), 'outer_control': control}
        sealed = None
        try:
            sealed = capsule(roots, capsule_root, durable, config['limits'])
            result['capsule'] = sealed
        except BaseException as error:
            result['capsule'] = {'status': 'FAIL_RAW_PRESERVATION_CLOSURE', 'error_type': type(error).__name__,
                                 'error': str(error), 'raw_roots_preserved': {key: str(value) for key, value in roots.items()},
                                 'partial_archive_preserved': str(capsule_root / 'PRIVATE-RAW-CAPSULE.zip'),
                                 'durable_partial_copy_if_present_preserved': str(durable / 'PRIVATE-RAW-CAPSULE.zip.partial'),
                                 'public_suitable': False, 'historical_raw_loss_closed': False}
            finalization_failure('capsule_preservation', error)
        if sealed is not None:
            for path, stage in ((capsule_root / 'CAPSULE-RECEIPT.json', 'temporary_capsule_receipt'),
                                (durable / 'CAPSULE-RECEIPT.json', 'durable_capsule_receipt')):
                try:
                    write_new(path, sealed)
                except BaseException as error:
                    finalization_failure(stage, error)
            if sealed['durable_delivery']['status'] != 'PASS_EXACT_DURABLE_PRIVATE_CAPSULE':
                error = CapacityBlocker('Complete raw capsule preserved in /tmp; workspace cannot hold its verified copy plus reserve')
                finalization_failure('durable_capsule_delivery', error)
                if result['status'] == 'FAIL_REQUIRED_FINALIZATION':
                    result['status'] = 'BLOCKED_DURABLE_CAPSULE_DELIVERY'
        # The temporary record is provisional, so a denied durable write
        # cannot leave a premature PASS. The durable record is authoritative
        # only after all required preservation and provisional records pass.
        provisional = {**result, 'status': 'PENDING_REQUIRED_DURABLE_FINALIZATION',
                       'observed_action_outcome': result['status'], 'authoritative_result': str(durable / 'RESULT.json')}
        try:
            write_new(capsule_root / 'RESULT.json', provisional)
        except BaseException as error:
            finalization_failure('temporary_result', error)
        try:
            write_new(durable / 'RESULT.json', result)
        except BaseException as error:
            finalization_failure('durable_result', error)
        if result.get('finalization_errors'):
            for path in (capsule_root / 'FINALIZATION-FAILURES.json', durable / 'FINALIZATION-FAILURES.json'):
                try:
                    write_new(path, {'status': 'FAIL_REQUIRED_FINALIZATION',
                        'primary_error_type': type(primary).__name__ if primary else None,
                        'primary_error': str(primary) if primary else None,
                        'finalization_errors': result['finalization_errors'],
                        'result_status': result['status'], 'raw_roots_preserved': {key: str(value) for key, value in roots.items()},
                        'platform_qualified': False, 'historical_raw_loss_closed': False})
                except BaseException as secondary:
                    if primary is not None:
                        primary.add_note('Secondary finalization-failure report: ' + str(secondary))
    print(json.dumps({'status': result['status'], 'completed': len(completed), 'result': str(durable / 'RESULT.json')}))
    if primary is not None:
        raise primary
