def sha(path):
 with Path(path).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def write(path,value):

def tree(root):
 result={}
 def denied(error):raise error
 for parent,dirs,files in os.walk(root,followlinks=False,onerror=denied):
  for name in sorted(dirs+files):
   p=Path(parent)/name;s=p.lstat();key=p.relative_to(root).as_posix()
   if stat.S_ISDIR(s.st_mode):row={'kind':'directory'}
   elif stat.S_ISREG(s.st_mode):row={'kind':'file','bytes':s.st_size,'sha256':sha(p)}
   elif stat.S_ISLNK(s.st_mode):row={'kind':'symlink','target':os.readlink(p)}
   else:raise ValueError('special evidence/input path')
   result[key]=row
 return result

def preserve_capsule(out,scratch):

  archive=out/'complete-runtime-evidence.zip'
  with zipfile.ZipFile(archive,'x',compression=zipfile.ZIP_DEFLATED,compresslevel=9) as z:
   for key,row in sorted(raw.items()):
