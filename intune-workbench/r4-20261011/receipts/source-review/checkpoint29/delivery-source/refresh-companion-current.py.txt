def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def require(ok, detail):
    if not ok:
        raise ValueError(detail)


def safe_relative(name):
    require(isinstance(name,str) and name and '\\' not in name, 'Unsafe relative path')
    path = PurePosixPath(name)
    require(not path.is_absolute() and all(x not in ('','.','..') for x in name.split('/')), 'Unsafe relative path')
    return name


def safe_file(path, limit=8*1024*1024):
    path = Path(path).absolute()
    for parent in [path,*path.parents]:
        require(not parent.is_symlink(), 'Symlink ancestor: '+str(parent))
    require(stat.S_ISREG(path.lstat().st_mode), 'Non-regular file: '+str(path))
    require(path.stat().st_size <= limit, 'File size bound: '+str(path))
    return path.read_bytes()


def dump(path, value):
    path = Path(path)
    path.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
    with path.open('xb') as out:
        out.write(json.dumps(value, indent=2, sort_keys=True).encode()+b'\n')
    path.chmod(0o600)


def write(path, raw):
    path.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
    with path.open('xb') as out:
        out.write(raw)
    path.chmod(0o600)


def raw_git(commit, path):
    git = Path('/usr/bin/git')
    require(git.is_file() and digest(git.read_bytes()) == '356db14e102d68a1a37d8a1ac577dfd678d45d46e92f468bef8b7154e7bfdc60', 'Git pin differs')
    argv = [str(git),'--no-optional-locks','-c','core.hooksPath=/dev/null','-c','core.fsmonitor=false',
            '-c','core.autocrlf=false','cat-file','blob',commit+':'+path]
    completed = subprocess.run(argv,cwd=REPO,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=30,
        env={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8','LC_ALL':'C.UTF-8','GIT_CONFIG_NOSYSTEM':'1',
             'GIT_CONFIG_GLOBAL':'/dev/null','GIT_CONFIG_SYSTEM':'/dev/null','GIT_NO_LAZY_FETCH':'1',
             'GIT_TERMINAL_PROMPT':'0','GIT_ALLOW_PROTOCOL':''})
    require(completed.returncode==0, 'Exact committed source blob unavailable: '+path)
    return completed.stdout


def public_files():
    root = REPO / META
    found = []
    def denied(error):
        raise error
    for parent, dirs, files in os.walk(root,followlinks=False,onerror=denied):
        for name in dirs:
            require(stat.S_ISDIR((Path(parent)/name).lstat().st_mode), 'Unsupported metadata directory type')
        for name in files:
            path = Path(parent)/name
            relative = safe_relative(path.relative_to(root).as_posix())
            require(relative.endswith(('.json','.md','.py.txt')), 'Unselected metadata extension: '+relative)
            # Every source text is checked against the actual latest reviewed
            # delta or exact previous public-admitted bytes before it is copied.
            found.append((relative,path))
    require(1 <= len(found) <= 100, 'Bounded public metadata census failed')
    return sorted(found)

def final_inputs(path, expected_sha, checkpoint):

            'Original successful C27 commit receipt required')

def admitted_prior(path, expected_sha, review_path, review_sha):
    raw=safe_file(path);review_raw=safe_file(review_path)
    require(digest(raw)==expected_sha and digest(review_raw)==review_sha, 'Prior selection/admission substituted')
    value=json.loads(raw);review=json.loads(review_raw)
    require(value.get('base_commit')=='3bf60e3ed150ec1d147ef2c07b7bea88f7ad480e'
            and value.get('source_commit')==ENGINE, 'Prior engine/base identity differs')
    require(review.get('selection_sha256')==expected_sha and review.get('source_commit')==ENGINE
            and review.get('documentation_commit')==value.get('documentation_commit'), 'Exact prior public admission linkage absent')
    tag=re.fullmatch(r'public-independent-companion-C([0-9]+)-stage-admission/1',review.get('schema') or '')
    require(tag is not None and int(tag.group(1))>=27
            and review.get('status')==f'PASS_SCOPED_FIXED_C{tag.group(1)}_SELECTION', 'Exact checkpoint-specific public prior admission absent')
    if 'checkpoint' in value:
        require(value['checkpoint']==int(tag.group(1)), 'Prior manifest/admission checkpoint differs')
    require(review.get('companion_base')==value['base_commit'] and review.get('companion_prefix')==value.get('prefix'),
            'Prior admitted companion namespace differs')
    rows={safe_relative(row['path']):row for row in value['files']}
    admitted={safe_relative(row['path']):row for row in review['files']}
    require(1 <= len(rows) <= 200 and len(rows)==len(value['files'])==review.get('selected_files')==len(admitted)
            and set(rows)==set(admitted), 'Prior exact admitted file set differs')
    for relative,row in rows.items():
        require((row['bytes'],row['sha256'])==(admitted[relative]['bytes'],admitted[relative]['sha256']), 'Prior selected member pin differs')
        source=Path(row['physical_path'])
        require(not source.is_symlink() and source.is_file() and source.stat().st_size==row['bytes'], 'Prior selected member type/size differs')
        if not relative.startswith('packages/'):
            require(digest(safe_file(source))==row['sha256'], 'Prior selected public text bytes changed')
    return value,raw,review,review_raw,rows,int(tag.group(1))


def main():

    parser.add_argument('--output',type=Path,help='NEW immediate stage beneath this owned directory')

    parser.add_argument('--prior-selection-sha256',default=PRIOR_SHA)
    parser.add_argument('--prior-admission',type=Path,default=PRIOR_ADMISSION)
    parser.add_argument('--prior-admission-sha256',default=PRIOR_ADMISSION_SHA)
    args=parser.parse_args()
    value,receipts=final_inputs(args.final_inputs,args.final_inputs_sha256,args.checkpoint)
    if args.validate_only:
        require(args.output is None, 'Validate-only does not accept stage output')
        print(json.dumps({'status':'PASS_SCOPED_RECEIPT_READER_ONLY','checkpoint':args.checkpoint,
                          'commit':value['documentation_commit'],'receipt_roles':sorted(receipts),
                          'product_or_native_execution':0,'prefix_modified':False}))
        return
    require(args.output is not None, 'New stage output required')
    output=args.output.absolute()
    require(output.parent==BASE and not output.exists() and not output.is_symlink(), 'New immediate owned stage directory required')
    require(not DEST.exists() and not any(p.is_symlink() for p in [DEST,*DEST.parents]),
            'Coordinator isolated prefix must remain fresh with no symlink ancestors')
    prior,prior_raw,prior_review,prior_review_raw,prior_rows,previous_checkpoint=admitted_prior(
        args.prior_selection,args.prior_selection_sha256,args.prior_admission,args.prior_admission_sha256)
    # Prior independent admission rehashed the exact ZIPs. This stage verifies
    # type/size and reuses that clearance without copying/rebuilding any ZIP.
    commit=value['documentation_commit'];checkpoint=value['checkpoint']
    covered={row['repository_path']:row for row in receipts['diff_review'][1]['exact_file_list_and_pins']}
    old_hashes={row['sha256'] for row in prior_rows.values()}
    output.mkdir(mode=0o700)
    payload=output/'payload';payload.mkdir(mode=0o700)
    history=output/'prior-admitted-revision';history.mkdir(mode=0o700)
    write(history/'SELECTION.json',prior_raw)
    write(history/'PUBLIC-STAGE-REVIEW.json',prior_review_raw)
    entries={name:dict(row,final_destination=str(DEST/name),action='REUSE_EXACT_PREVIOUS_ADMITTED_SELECTION') for name,row in prior_rows.items()}
    changes=[]
    def stage(origin,relative,classification,source_path=None,data=None):
        relative=safe_relative(relative)
        raw=safe_file(origin) if data is None else data
        raw.decode('utf-8')
        require(len(raw)<=8*1024*1024, 'Selected text size bound')
        reuse=None
        if source_path:
            require(raw_git(commit,source_path)==raw, 'Selected source differs from exact final Git blob: '+source_path)
            require(source_path in covered or digest(raw) in old_hashes, 'Selected source path lacks exact reviewed delta/reuse: '+source_path)
            if source_path in covered:
                require(covered[source_path]['sha256']==digest(raw) and covered[source_path]['bytes']==len(raw), 'Reviewed delta pin mismatch')
                reuse='LATEST_EXACT_PUBLIC_DELTA_AND_GIT_BLOB'
            else:
                reuse='PREVIOUS_ADMITTED_EXACT_BYTES_AND_LATEST_GIT_BLOB'
        target=payload/relative
        write(target,raw)
        old=prior_rows.get(relative)
        if old and old['sha256']!=digest(raw):
            before=safe_file(old['physical_path'])
            require((len(before),digest(before))==(old['bytes'],old['sha256']), 'Previous changed bytes differ before preservation')
            write(history/relative,before)
            changes.append({'path':relative,'before_bytes':len(before),'before_sha256':digest(before),
                            'after_bytes':len(raw),'after_sha256':digest(raw),'prior_bytes_preserved':str(history/relative)})
        entries[relative]={'path':relative,'physical_path':str(target),'final_destination':str(DEST/relative),
                          'bytes':len(raw),'sha256':digest(raw),'origin':str(origin),'classification':classification,
                          'action':'STAGED_TEXT_REPLACEMENT' if old else 'STAGED_TEXT_ADDITION',
                          'observed_documentation_commit':commit if source_path else None,'source_path':source_path,
                          'evidence_reuse':reuse}
    for name in DOCS:
        stage(REPO/'docs'/name,'work/docs/'+name,'exact_latest_reviewed_documentation','docs/'+name)
    found=public_files()
    for relative,path in found:
        raw=safe_file(path)
        stage(path,'receipts/source-review/'+relative,'public_selected_metadata_or_exact_qualification_source_text',META+'/'+relative,raw)
    helper=REPO/META/'native-recovery-plan-qualifier.py.txt'
    stage(helper,'tools/native_recovery_plan.py','optional_exact_qualification_text_not_standalone_bundle',META+'/native-recovery-plan-qualifier.py.txt')
    role_names={'commit':f'COMMIT-{checkpoint}.json','diff_review':f'DOCUMENTATION-REVIEW-{checkpoint}.json',
                'commit_binding':f'DOCUMENTATION-COMMIT-BINDING-{checkpoint}.json','source_readback':f'SOURCE-DOCUMENTATION-READBACK-{checkpoint}.json',
                'branch_publication':f'SOURCE-BRANCH-READBACK-{checkpoint}.json'}
    for role,(row,value,raw) in receipts.items():
        stage(row['path'],'receipts/'+role_names[role],'bounded_final_commit_review_readback_metadata_pending_companion_review',data=raw)
