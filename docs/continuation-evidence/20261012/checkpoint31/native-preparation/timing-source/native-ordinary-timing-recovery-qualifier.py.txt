def _exact_assignments(actual, expected):
 # Assignment order is a declared set by ID; duplicate IDs never collapse.

  if type(rows) is not list:return False

def run(action,output,*,previous_root,seed_root):
 os.umask(0o077)
 root,previous,seed,route=_arguments(action,output,previous_root=previous_root,seed_root=seed_root)
 root.mkdir(mode=0o700)
 if route=='known-completion':result=_known_completion(root,previous,seed)
 else:
  phase='explicit_fresh_lease_release'
  try:
   released=_release(root/'release',seed=previous,preparation=seed);assert released['status']=='PASS'
   phase='new_reviewed_native_plan'
   prepared=_prepare(root/'preparation',root/'release',preparation=seed,original=previous,route=route);assert prepared['status']=='PASS'
   phase='separately_signed_successor_transfer_and_apply'
   result=_successor(root,seed=previous,prep=root/'preparation',release=root/'release',route=route)
   result.update(source_revision=Q.source_revision,historical_source_revision=json.loads((previous/'manifest.json').read_bytes())['run_context']['source_revision'],route=route)
   write_json(root/'receipt.json',result)
  except BaseException as error:
   _record_failure(root/'recovery-failure.json',{'status':'FAIL','phase':phase,'error':getattr(error,'code',type(error).__name__),'platform_qualified':False,'production_approved':False},error)
   raise
 if type(result) is not dict or result.get('status')!='PASS':raise ValueError('Actual timing recovery did not qualify')
 from native_ordinary_linkage import artifact_census,canonical as closed_canonical
 link={'schema_version':'ordinary-native-timing-recovery-linkage/1','action':action,'route':route,'source_revision':Q.source_revision,
       'historical_source_revision':result['historical_source_revision'],'previous_root':str(previous),'seed_root':str(seed),
       **{key:result[key] for key in ('operation_id','manifest_sha256','job_root','policy_path')},'artifacts':artifact_census(root),
       'scope':'Selected outputs only; full new receiving/job/authority/history/private raw evidence and original layers require coordinator typed capsule.'}
 target=root/'qualification-linkage.json';fd=os.open(target,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
 with os.fdopen(fd,'wb') as stream:stream.write(closed_canonical(link));stream.flush();os.fsync(stream.fileno())
 return {**result,'qualification_linkage':link}
