class ProfileError(ValueError):
    pass


def require(condition, message):
    if not condition:
        raise ProfileError(message)


def canonical(value):
    def closed(item, depth=0):
        require(depth <= 64, 'qualification JSON nesting exceeded')
        if type(item) in (str, int, bool) or item is None:
            return
        if type(item) is float:
            require(math.isfinite(item), 'nonfinite qualification JSON denied')
        elif type(item) is list:
            for child in item:
                closed(child, depth + 1)
        elif type(item) is dict:
            require(all(type(key) is str for key in item), 'qualification JSON object keys must be strings')
            for child in item.values():
                closed(child, depth + 1)
        else:
            raise ValueError('unsupported qualification JSON value')
    closed(value)
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False).encode()


def path_under(root, name):
    require(type(name) is str and '\\' not in name and '\x00' not in name, 'invalid relative path')
    rel = PurePosixPath(name)
    require(not rel.is_absolute() and name == str(rel) and all(p not in ('', '.', '..') for p in rel.parts), 'invalid relative path')
    result = root / name
    require(not any(p.is_symlink() for p in (result, *result.parents)), 'symlink input denied')
    return result


def root_path(value):
    path = Path(value).absolute()
    require('..' not in path.parts and not any(p.is_symlink() for p in (path, *path.parents)), 'unsafe root')
    require(path.is_dir(), 'missing root')
    return path


def hash_file(path, maximum=MAX_FILE):
    info = path.lstat()
    require(stat.S_ISREG(info.st_mode) and 0 <= info.st_size <= maximum, 'unsupported or oversized input')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    with os.fdopen(fd, 'rb') as stream:
        before = os.fstat(stream.fileno())
        value = hashlib.file_digest(stream, 'sha256').hexdigest()
        after = os.fstat(stream.fileno())
    identity = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns)
    require(identity(info) == identity(before) == identity(after) == identity(path.lstat()), 'input changed during verification')
    return {'bytes': info.st_size, 'sha256': value}


def inventory(root):
    rows = {}
    def traversal_failed(error):
        raise ProfileError('input tree traversal failed: ' + str(error)) from error
    for current, directories, files in os.walk(root, followlinks=False, onerror=traversal_failed):
        for name in directories:
            require(not (Path(current) / name).is_symlink(), 'directory symlink denied')
        for name in files:
            path = Path(current) / name
            relative = path.relative_to(root).as_posix()
            require(len(rows) < MAX_FILES, 'file count bound')
            rows[relative] = hash_file(path)
    require(rows, 'empty input tree')
    return dict(sorted(rows.items()))


def load_json(path, maximum=16 * 1024 * 1024):
    hash_file(path, maximum)
    def pairs(rows):
        result = {}
        for key, value in rows:
            require(key not in result, 'duplicate JSON key')
            result[key] = value
        return result
    return json.loads(path.read_bytes(), object_pairs_hook=pairs)

    def input(self, name):
        path = 'inputs/' + name
        require(path in self.value['inputs'], 'unlisted fixture input')
        return path_under(self.evidence, path)

    def tool(self, name):
        require(name in self.paths, 'unlisted tool')
        return str(self.paths[name])

    def assert_import_origins(self):
        origins = {}
        for name, module in tuple(sys.modules.items()):
            if name == 'intune_iac' or name.startswith('intune_iac.') or name == 'plugin_tests' or name.startswith('plugin_tests.'):
                source = getattr(module, '__file__', None)
                require(source is not None, 'package module has no source origin')
                path = Path(source).absolute()
                require(self.project in path.parents and path.suffix == '.py', 'package module imported outside selected package')
                relative = path.relative_to(self.project).as_posix()
                require(relative in self.value['package'] and hash_file(path) == self.value['package'][relative], 'imported package module changed')
                origins[name] = {'path': relative, **self.value['package'][relative]}
            elif name in {Path(key).stem for key in self.value['helpers'] if key.endswith('.py')}:
                source=getattr(module,'__file__',None)
                require(source is not None,'helper module has no source origin')
                path=Path(source).absolute();key='helpers/'+name+'.py'
                require(path==self.evidence/key and key in self.value['helpers'] and hash_file(path)==self.value['helpers'][key],
                    'helper imported outside exact selected bundle')
                origins[name]={'external_helper':key,**self.value['helpers'][key]}
        return origins

    def recheck(self):
        require(inventory(self.project) == self.value['package'], 'package changed during qualification')
        require(self.value['helpers'] == {'helpers/' + name: row for name, row in inventory(self.evidence / 'helpers').items()}, 'helper file set changed during qualification')
        require(self.value['inputs'] == {'inputs/' + name: row for name, row in inventory(self.evidence / 'inputs').items()}, 'input file set changed during qualification')
        for group in ('helpers', 'inputs'):
            for name, expected in self.value[group].items():
                require(hash_file(path_under(self.evidence, name)) == expected, 'qualification dependency changed')
        for role, path in self.paths.items():
            require(hash_file(path) == self.value['tools'][role], 'tool changed during qualification')
        require(hash_file(Path(__file__).absolute())['sha256'] == self.value['loader_sha256'], 'loader changed during qualification')
        return self.assert_import_origins()

def typed_original_inventory(root):
    root = root_path(root); rows={}; pending=[root]
    while pending:
        path=pending.pop(); info=path.lstat(); name=path.relative_to(root).as_posix()
        require(len(rows)<16000,'bounded original typed tree')
        if stat.S_ISDIR(info.st_mode):
            rows[name]={'kind':'directory','mode':stat.S_IMODE(info.st_mode)}
            with os.scandir(path) as entries: pending.extend(sorted((Path(row.path) for row in entries),reverse=True))
        elif stat.S_ISREG(info.st_mode): rows[name]={'kind':'file','mode':stat.S_IMODE(info.st_mode),**hash_file(path)}
        elif stat.S_ISLNK(info.st_mode): rows[name]={'kind':'symlink','literal_target':os.readlink(path)}
        elif stat.S_ISSOCK(info.st_mode): rows[name]={'kind':'runtime_socket_metadata','mode':stat.S_IMODE(info.st_mode),'restored_connection':False}
        else: raise ProfileError('Unsupported original special node retained; no closure or dispatch')
    return dict(sorted(rows.items()))

def authority_snapshot(root, value, facts):
    import sqlite3
    require(type(value) is dict and set(value)=={'snapshots','job','owners','consumed','original_before_after_unchanged','copy_precedes_SQLite_open'}
        and value['copy_precedes_SQLite_open'] is True and value['original_before_after_unchanged'] is True,
        'copied original authority reader required')
    snapshots=value['snapshots']; require(type(snapshots) is list and len(snapshots)==2,'both service and broker snapshots required')
    required=('trust/authority/service.sqlite3','jobs/approved/broker.sqlite3'); copies={}
    for snapshot,relative in zip(snapshots,required):
        require(type(snapshot) is dict and set(snapshot)=={'relative','files'} and snapshot['relative']==relative
            and type(snapshot['files']) is list and len(snapshot['files'])==3,'closed original database/WAL/SHM snapshot')
        for selected,suffix in zip(snapshot['files'],('','-wal','-shm')):
            require(type(selected) is dict and set(selected)=={'original','copied','present'} and type(selected['present']) is bool,
                'typed snapshot fields required')
            original=path_under(root,relative+suffix)
            if selected['present']:
                require(type(selected['original']) is dict and selected['original']['path']==str(original)
                    and canonical(hash_file(original))==canonical({key:selected['original'][key] for key in ('bytes','sha256')}),
                    'current original database differs from admitted before snapshot')
                copy_pin=selected['copied']; copied=Path(copy_pin['path'])
                require(root!=copied and root not in copied.parents and root not in copied.parent.parents
                    and canonical(hash_file(copied))==canonical({key:copy_pin[key] for key in ('bytes','sha256')})
                    and canonical({key:selected['original'][key] for key in ('bytes','sha256')})
                    ==canonical({key:copy_pin[key] for key in ('bytes','sha256')}),'actual complete copied database member differs')
                require(not any(p.is_symlink() for p in (copied,*copied.parents)),'copied authority must be literal')
                if suffix=='': copies[relative]=copied
                else: require(copied==Path(str(copies[relative])+suffix),'WAL/SHM copied layout must match actual SQLite receiver')
            else: require(suffix!='' and selected['original'] is None and selected['copied'] is None
                and not original.exists() and not original.is_symlink(),'absent database member must really be absent')
    with sqlite3.connect(copies[required[0]].as_uri()+'?mode=ro',uri=True) as db:
        db.execute('PRAGMA query_only=ON')
        job=db.execute('SELECT id,root,manifest_sha256,status,dispatches,outcome,controller_pid FROM jobs WHERE id=?',(facts['operation_id'],)).fetchone()
        owners=db.execute('SELECT scope,job FROM ownership WHERE job=? ORDER BY scope',(facts['operation_id'],)).fetchall()
        consumed=db.execute('SELECT operation_id,manifest_sha256 FROM consumed WHERE operation_id=?',(facts['operation_id'],)).fetchall()
    require(job is not None and job[:5]==(facts['operation_id'],facts['job_root'],facts['manifest_sha256'],facts['status'],1)
        and type(job[4]) is int and job[6] is None and type(job[5]) is str,'actual before authority status/consumption/host must match route')
    manifest=facts['manifest']; physical=manifest['provider_binding']['backend']['state_blob']
    expected=sorted([hashlib.sha256(canonical({'tenant':manifest['tenant_id'],'object_and_assignments':oid})).hexdigest()
        for oid in manifest['object_ids']]+[hashlib.sha256(canonical({'tenant':manifest['tenant_id'],'state_blob':physical})).hexdigest()])
    require(len(expected)==2 and owners==[(scope,facts['operation_id']) for scope in expected]
        and consumed==[(facts['operation_id'],facts['manifest_sha256'])],'exact two writer reservations and one consumed original required')
    observed={'operation_id':job[0],'job_root':job[1],'manifest_sha256':job[2],'status':job[3],'dispatches':job[4],
        'outcome_sha256':hashlib.sha256(job[5].encode()).hexdigest(),'controller_pid':job[6]}
    require(canonical(value['job'])==canonical(observed) and canonical(value['owners'])==canonical([list(row) for row in owners])
        and canonical(value['consumed'])==canonical([list(row) for row in consumed]),'typed witnessed SQL rows differ from copied actual authority')
    return observed
