#!/usr/bin/env python3
"""Bind existing exact Linux validation fixtures, without installing or running them.

This is validation preparation, not protected execution, CI dispatch, native
host qualification or authority to fetch/provision binaries. The fixture
custodian must supply the existing pinned local inputs on an approved runner.
"""
import argparse
import importlib.util
import json
import os
from pathlib import Path
import sys

import qualification_tools as q

HERE = Path(__file__).resolve().parent
LEGACY_ENV = 'INTUNE_TEST_TOFU_EXECUTABLE'


def preparation_api():
    spec = importlib.util.spec_from_file_location('intune_ci_fixture_preparation_existing',
                                                HERE / 'prepare-qualification-tools.py')
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def environment(paths):
    # Four explicit local paths; absent input may never fall back to a workspace
    # namespace on the CI host. Consumers independently enforce fixed hashes.
    keys = {item['environment'] for item in q.PINNED.values()}
    q.require(type(paths) is dict and set(paths) == keys,
              'All four explicit qualification input paths are required')
    q.require(all(type(value) is str and value for value in paths.values()),
              'Empty or nonliteral qualification input is forbidden')
    records = q.records(paths)
    result = {item['environment']: item['path'] for item in records.values()}
    result[LEGACY_ENV] = records['tofu']['path']
    return result, records


def prepare(output, paths):
    values, records = environment(paths)
    api = preparation_api()
    root, fd = api.new_directory(output)
    receipt = {'schema': 'intune-ci-fixed-fixture-bindings/1', 'status': 'RUNNING',
               'tools': records, 'environment': values, 'binary_execution': False,
               'network_attempted': False, 'installed': False,
               'legacy_path_created': False, 'tool_pins_changed': False,
               'ci_qualified': False, 'production_approved': False}
    try:
        checked, again = environment({name: values[name] for name in paths})
        q.require(checked == values and again == records,
                  'Qualification inputs changed during preparation')
        api.guard(root, fd)
        api.document(fd, 'environment.json', values)
        receipt['status'] = 'PASS_FIXED_CI_INPUT_BINDINGS'
        api.document(fd, 'receipt.json', receipt)
        return receipt
    except BaseException as error:
        receipt.update(status='FAIL_PRESERVED', error_type=type(error).__name__)
        try:
            api.document(fd, 'failure.json', receipt)
        except BaseException as secondary:
            print(json.dumps({'secondary_receipt_failure': type(secondary).__name__}),
                  file=sys.stderr)
        raise
    finally:
        os.close(fd)


def export(input_root):
    # Export only a small path-only map. Never source shell text, emit caller
    # keys/hashes/secrets or create an ambiguous historical /tmp tool alias.
    root = q.absolute(input_root)
    with q.opened(root / 'environment.json', maximum=32768) as (stream, _):
        raw = stream.read(32769)
    def unique(pairs):
        result = {}
        for key, value in pairs:
            q.require(key not in result, 'Duplicate environment key')
            result[key] = value
        return result
    values = json.loads(raw, object_pairs_hook=unique)
    keys = {item['environment'] for item in q.PINNED.values()}
    q.require(type(values) is dict and set(values) == keys | {LEGACY_ENV},
              'Unexpected qualification environment fields')
    checked, _ = environment({name: values[name] for name in keys})
    q.require(values == checked, 'Native fixture path differs from pinned Tofu')
    # q.absolute rejects newline/control characters before this text is emitted.
    return ''.join(name + '=' + checked[name] + '\n' for name in sorted(checked))


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    commands = parser.add_subparsers(dest='command', required=True)
    binding = commands.add_parser('bind')
    binding.add_argument('--output', required=True)
    for role in q.PINNED:
        binding.add_argument('--' + role.replace('_', '-'), required=True)
    exporting = commands.add_parser('export')
    exporting.add_argument('--input', required=True)
    args = parser.parse_args(argv)
    try:
        if args.command == 'export':
            sys.stdout.write(export(args.input))
        else:
            paths = {item['environment']: getattr(args, role)
                     for role, item in q.PINNED.items()}
            print(json.dumps(prepare(args.output, paths), sort_keys=True))
        return 0
    except (OSError, ValueError, TypeError) as error:
        print(json.dumps({'status': 'BLOCKED', 'error_type': type(error).__name__,
                          'preserve_partial_output': True}), file=sys.stderr)
        return 2


if __name__ == '__main__':
    raise SystemExit(main())
