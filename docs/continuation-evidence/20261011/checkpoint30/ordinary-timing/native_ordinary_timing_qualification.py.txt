def _record_failure(root,receipt,primary):
    """Archive failure without bypassing cleanup or replacing its primary."""
    try:write_json(Path(root)/'receipt.json',receipt)
    except BaseException as secondary:
        if isinstance(secondary,(KeyboardInterrupt,SystemExit)):
            secondary.__cause__=primary
            return secondary
        primary.add_note('Secondary failure receipt recording: '+str(getattr(secondary,'code',type(secondary).__name__)))
    return primary

def _typed_literal_equal(actual,expected):
    if type(actual) is not type(expected):return False
    if type(actual) is dict:
        return set(actual)==set(expected) and all(_typed_literal_equal(actual[key],expected[key]) for key in expected)
    if type(actual) is list:
        return len(actual)==len(expected) and all(_typed_literal_equal(a,b) for a,b in zip(actual,expected))
    return actual==expected


def _raw_assignments_equal(actual,expected):
    if type(actual) is not list or type(expected) is not list or len(actual)!=len(expected):return False
    if any(type(row) is not dict or type(row.get('id')) is not str for row in actual+expected):return False
    a={row['id']:row for row in actual};b={row['id']:row for row in expected}
    return len(a)==len(actual) and len(b)==len(expected) and _typed_literal_equal(a,b)

def _authority_rows(policy_path):
    value=json.loads(Path(policy_path).read_bytes());path=Path(value['authority_root'])/'service.sqlite3'
    with sqlite3.connect('file:'+str(path)+'?mode=ro',uri=True) as db:
        names=[r[0] for r in db.execute("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'")]
        return {name:[list(row) for row in db.execute('SELECT * FROM "'+name.replace('"','""')+'" ORDER BY rowid')] for name in names}


def _job_files(root):
    result={}
    def denied(error):raise error
    for parent,dirs,names in os.walk(root,followlinks=False,onerror=denied):
        for name in dirs+names:assert not (Path(parent)/name).is_symlink()
        for name in names:
            path=Path(parent)/name;result[path.relative_to(root).as_posix()]={'bytes':path.stat().st_size,'sha256':support.sha(path.read_bytes())}
    return result


def restore(output,previous_root,seed_root):
    root=Path(output);root.mkdir(mode=0o700);previous=Path(previous_root);seed=Path(seed_root)
    # Preparation copy is explicit and records its source. Actual restoration
    # then reopens those exact prepared receiving bytes under a NEW approval.
    prep_root=root/'preparation';prep_root.mkdir(mode=0o700)
    selected=json.loads((previous/'qualification-linkage.json').read_bytes());old_rows=_authority_rows(selected['policy_path']);old_files=_job_files(Path(selected['job_root']))
    write_json(root/'previous-authority-rows.json',old_rows);write_json(root/'previous-job-files.json',old_files)
    fixture=support.copied_fixture(previous,prep_root);handles=support.new_handles();primary=None
    try:
        prior=json.loads((previous/'receipt.json').read_bytes());assert prior['status']=='PASS'
        assert fixture.graph['policy']['description']=='Fixture intentionally corrected'
        assert fixture.blobs[STATE_PATH]['lease'] is None and not fixture.blobs[STATE_PATH]['metadata']
        adapter=support.rebind_adapter(support.load_adapter(seed/'prepared'),fixture.state_bytes(),description='Fixture only')
        planned=support.plan_preparation(prep_root,fixture,adapter,handles,expected_exit=2)
        assert not json.loads(planned['plan_json']).get('resource_drift')
        assert planned['receipt']['saved_plan_sha256']!=prior['saved_plan_sha256']
        write_json(root/'restoration-review.json',{'new_reviewed_plan_sha256':planned['receipt']['saved_plan_sha256'],'previous_plan_sha256':prior['saved_plan_sha256'],'before':'Fixture intentionally corrected','after':'Fixture only','new_approval_required':True,'same_authority':json.loads((previous/'qualification-linkage.json').read_bytes())['policy_path']})
    except BaseException as error:primary=error
    support.close_all(prep_root,primary,handles,fixture)
    result=apply(root,seed,prepared_dir=prep_root/'prepared',fixture_root=prep_root,reuse_authority=previous)
    rows=_authority_rows(selected['policy_path']);files=_job_files(Path(selected['job_root']))
    checks={'prior_authority_rows_preserved':all(name in rows and all(canonical(row) in [canonical(item) for item in rows[name]] for row in prior_rows) for name,prior_rows in old_rows.items()),'prior_job_files_preserved':files==old_files,'new_restoration_operation':result['operation_id']!=selected['operation_id']}
    write_json(root/'previous-history-after.json',{'rows':rows,'files':files,'checks':checks})
    result['restoration_history_assertions']=checks
    if not all(checks.values()):result['status']='FAIL'
    write_json(root/'receipt.json',result)
    assert all(checks.values()),checks
    return result


def _emit_timing_linkage(action,output,result,previous,seed,recording_mode):
    from native_ordinary_linkage import literal_path,pin,load_json,canonical as closed_canonical,artifact_census
    if type(result) is not dict or result.get('status')!='PASS':raise ValueError('Passing actual timing receipt required')
    root=literal_path(output);policy=literal_path(result['policy_path']);job=literal_path(result['job_root'])
    manifest=load_json(job/'manifest.json');receipt=load_json(root/'receipt.json')
    if manifest['operation_id']!=result['operation_id'] or digest(manifest)!=result['manifest_sha256']:raise ValueError('Actual selected timing manifest differs')
    if closed_canonical(receipt)!=closed_canonical(result):raise ValueError('Exact returned timing receipt differs')
    for path in (job/'manifest.json',job/'policy-binding.json',policy):pin(path)
    files=artifact_census(root)
    link={'schema_version':'ordinary-native-fault-timing-linkage/1','action':action,'recording_mode':recording_mode,
          'source_revision':Q.source_revision,'kind':'operation','seed_root':str(seed),'previous_root':str(previous),
          **{key:result[key] for key in ('operation_id','manifest_sha256','job_root','policy_path')},
          'receiving_fixture':str(root/'fixture'),'artifacts':files,
          'artifact_scope':'Selected immutable helper outputs only; full receiving/job/authority/host evidence requires the separately inventoried private owned capsule.'}
    target=root/'qualification-linkage.json'
    fd=os.open(target,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    with os.fdopen(fd,'wb') as stream:stream.write(closed_canonical(link));stream.flush();os.fsync(stream.fileno())
    if artifact_census(root)!=files:raise ValueError('Timing artifacts changed while linking')
    return {**result,'qualification_linkage':link}


def run(action, output, *, previous_root, seed_root, recording_mode='archive-interruption'):
    os.umask(0o077)
    choices={'ordinary-fault-acquisition':'acquisition','ordinary-fault-policy-response':'policy-response','ordinary-fault-apply-recording':'evidence-recording'}
    if action not in choices or recording_mode not in RECORDING_MODES:raise ValueError('Closed new timing action/mode required')
    stage=choices[action]
    if stage!='evidence-recording' and recording_mode!='archive-interruption':raise ValueError('Recording mode cannot alter cancellation action')
    from native_ordinary_linkage import literal_path,load_json
    output=literal_path(output);seed=literal_path(seed_root);previous=literal_path(previous_root)
    if previous!=seed or output.exists() or not seed.is_dir():raise ValueError('Fresh timing output and exact ordinary seed required')
    for protected in (seed,Q.project,Q.evidence):
        if output==protected or protected in output.parents or output in protected.parents:raise ValueError('Timing output overlaps protected input')
    receipt=load_json(seed/'receipt.json')
    if receipt.get('status')!='PASS' or receipt.get('kind')!='preparation':raise ValueError('Qualified exact ordinary preparation required')
    if not (seed/'prepared/approved.plan').is_file():raise ValueError('Exact reviewed native saved plan required')
    result=apply(output,seed,stage=stage,recording_mode=recording_mode)
    return _emit_timing_linkage(action,output,result,previous,seed,recording_mode)
